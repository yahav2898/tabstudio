    Map every static-graph entry target that can compile the project to the source-root target on each

  <ItemGroup Condition="'$(NETCoreSdkVersion)' != '' AND '$(IsGraphBuild)' == 'true'">
    <ProjectReferenceTargets Include="Build" Targets="$(_msCoverageExtensionSourceRootTargetName)" SkipNonexistentTargets="true" />

          DependsOnTargets="ResolveProjectReferences" >
    <!--
      Resolve the source-root mapping from each referenced project itself, rather than reusing this test
      project's source roots. Deterministic source-root prefixes (/_/, /_1/, ...) are assigned per project
      based on each project's own SourceRoot set and ordering, so a referenced project's /_/ can point to a
      different directory than this project's /_/. Using this project's mapping for all references produced
      empty coverage reports under ContinuousIntegrationBuild=true (deterministic source paths) whenever the
      mappings did not line up.
    -->
    <MSBuild Projects="@(AnnotatedProjects->'%(FullPath)')"
             Targets="$(_msCoverageExtensionSourceRootTargetName)"
             Properties="TargetFramework=%(AnnotatedProjects.NearestTargetFramework)"
             SkipNonexistentTargets="true">

              ItemName="_msCoverageExtensionLocalTopLevelSourceRoot" />

    <CompilerVisibleProperty Include="RefitGeneratedRequestBuilding"/>
    <CompilerVisibleProperty Include="RefitEmitGeneratedCodeMarkers"/>
    <CompilerVisibleProperty Include="DisableRefitSourceGenerator"/>

  <!--
  Opt-in generated-only build profile. Set RefitRequireGeneratedRequests to true in a project that publishes with
  Native AOT or registers clients generated-only, and every RF006 reflection fallback fails the build instead of
  surfacing as a runtime exception. It is off by default so compatibility builds that deliberately use
  Refit.Reflection are unaffected; 'dotnet_diagnostic.RF006.severity = error' in .editorconfig is the per-folder
  equivalent.

  This has to be a target: the project body that sets the property is evaluated after this .props file.
  -->
  <Target Name="RefitRequireGeneratedRequests"
          BeforeTargets="CoreCompile"
          Condition="'$(RefitRequireGeneratedRequests)' == 'true'">
    <PropertyGroup>
      <WarningsAsErrors>$(WarningsAsErrors);RF006</WarningsAsErrors>
    </PropertyGroup>
  </Target>

  <PropertyGroup>
    <!-- The Roslyn version of the single analyzers\dotnet\roslyn<version>\cs slot in this package. -->
    <_RefitMinimumCompilerVersion>4.8</_RefitMinimumCompilerVersion>
  </PropertyGroup>

  <!--
  Fail loudly on a compiler older than the shipped analyzer slot.

  NuGet asset selection drops an analyzer slot newer than the host compiler without reporting
  anything, so such a consumer would otherwise see no generated clients and a pile of unrelated
  "type or namespace not found" errors with no hint at the cause.

  This has to be a target rather than a plain Condition: $(CompilerApiVersion) is set by the Roslyn
  targets, which are imported long after this .props file is evaluated, so it is always empty here at
  evaluation time. Every guard below is deliberately fail-open - an unset, non-Roslyn or malformed
  value skips the check rather than breaking a build we do not understand.
  -->
  <Target Name="RefitVerifyCompilerVersion"
          BeforeTargets="CoreCompile"
          Condition="'$(DisableRefitSourceGenerator)' != 'true'
                     and '$(Language)' == 'C#'
                     and '$(CompilerApiVersion)' != ''
                     and $(CompilerApiVersion.StartsWith('roslyn'))">
    <PropertyGroup>
      <_RefitCompilerVersion>$(CompilerApiVersion.Substring(6))</_RefitCompilerVersion>
    </PropertyGroup>
    <Error Condition="$(_RefitCompilerVersion.Contains('.'))
                      and '$([System.Version]::Parse($(_RefitMinimumCompilerVersion)).CompareTo($([System.Version]::Parse($(_RefitCompilerVersion)))))' &gt; '0'"
           Code="REFIT001"
           Text="Refit's source generator requires Roslyn $(_RefitMinimumCompilerVersion) or later (Visual Studio 2022 17.8+, or .NET SDK 8.0.100+), but this project is building with Roslyn $(_RefitCompilerVersion). Upgrade the build tools, or set DisableRefitSourceGenerator to true and use the Refit.Reflection package, which builds requests at runtime instead."/>
  </Target>

</Project>
