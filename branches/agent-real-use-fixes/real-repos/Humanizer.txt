      Resolve the source-root mapping from each referenced project itself, rather than reusing this test
      project's source roots. Deterministic source-root prefixes (/_/, /_1/, ...) are assigned per project
      based on each project's own SourceRoot set and ordering, so a referenced project's /_/ can point to a
      different directory than this project's /_/. Using this project's mapping for all references produced
      empty coverage reports under ContinuousIntegrationBuild=true (deterministic source paths) whenever the
      mappings did not line up.

             Targets="$(_msCoverageExtensionSourceRootTargetName)"
             Properties="TargetFramework=%(AnnotatedProjects.NearestTargetFramework)"

              ItemName="_msCoverageExtensionLocalTopLevelSourceRoot" />

                      Overwrite="true" Encoding="Unicode"
                      Condition="'@(_msCoverageExtensionMapping)'!=''" />

      <FileWrites Include="$(_msCoverageExtensionSourceRootMappingFilePath)" Condition="'@(_msCoverageExtensionMapping)'!=''" />
    </ItemGroup>
