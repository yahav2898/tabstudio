cmake_minimum_required(VERSION 3.22)
project(tabstudio_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama)
add_library(tabstudio-ai SHARED bridge.cpp)
target_link_libraries(tabstudio-ai PRIVATE llama)
set_target_properties(tabstudio-ai PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(ANDROID)
  target_link_options(tabstudio-ai PRIVATE "-Wl,-z,max-page-size=16384")
endif()
add_executable(tabstudio-ai-bench bench.cpp)
target_link_libraries(tabstudio-ai-bench PRIVATE tabstudio-ai)
